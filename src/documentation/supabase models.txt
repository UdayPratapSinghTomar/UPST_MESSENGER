    id: {
      type: DataTypes.UUID,
      defaultValue: DataTypes.UUIDV4,
      primaryKey: true,
    },

    name: {
      type: DataTypes.TEXT,
      allowNull: false,
    },

      type: DataTypes.TEXT,
      allowNull: false,
      unique: true,
    },

      type: DataTypes.TEXT,
      allowNull: false,
    },

      type: DataTypes.BOOLEAN,
      allowNull: false,
      defaultValue: false,
    },

    suspended_at: DataTypes.DATE,

    suspended_by: {
      type: DataTypes.UUID,
      references: {
        model: 'profiles',
        key: 'id'
      }
    },

    suspension_reason: DataTypes.TEXT

  }, {
    tableName: 'organizations',
    timestamps: true,
    underscored: true,
    createdAt: 'created_at',
    updatedAt: 'updated_at',
  });

  return Organization;
};

module.exports = (models) => {
  const {
    Organization,
    Profile,
    UserRole,
    Project,
    Task,
    TaskAssignment,
    TaskNote,
    TaskURL,
    TaskAttachment,
    ProjectAttachment
  } = models;

  // =========================================================
  // ORGANIZATION ↔ PROFILE
  // =========================================================
  Organization.hasMany(Profile, { foreignKey: 'organization_id' });
  Profile.belongsTo(Organization, { foreignKey: 'organization_id' });

  // =========================================================
  // ORGANIZATION ↔ USER ROLES
  // =========================================================
  Organization.hasMany(UserRole, { foreignKey: 'organization_id' });
  UserRole.belongsTo(Organization, { foreignKey: 'organization_id' });

  // ⚠️ IMPORTANT:
  // user_roles.user_id → auth.users (NOT profiles)
  // So DO NOT create belongsToMany(Profile <-> Organization)

  // =========================================================
  // ORGANIZATION SUSPENSION
  // =========================================================
  Organization.belongsTo(Profile, {
    foreignKey: 'suspended_by',
    as: 'suspendedBy'
  });

  // =========================================================
  // PROJECT RELATIONS
  // =========================================================

  // Organization ↔ Project
  Organization.hasMany(Project, { foreignKey: 'organization_id' });
  Project.belongsTo(Organization, { foreignKey: 'organization_id' });

  // Profile ↔ Project
  Project.belongsTo(Profile, { foreignKey: 'user_id', as: 'owner' });
  Project.belongsTo(Profile, { foreignKey: 'created_by_user_id', as: 'createdBy' });
  Project.belongsTo(Profile, { foreignKey: 'assigned_user_id', as: 'assignedUser' });

  Profile.hasMany(Project, { foreignKey: 'user_id', as: 'ownedProjects' });
  Profile.hasMany(Project, { foreignKey: 'created_by_user_id', as: 'createdProjects' });
  Profile.hasMany(Project, { foreignKey: 'assigned_user_id', as: 'assignedProjects' });

  // Project ↔ Attachments
  Project.hasMany(ProjectAttachment, { foreignKey: 'project_id' });
  ProjectAttachment.belongsTo(Project, { foreignKey: 'project_id' });

  // =========================================================
  // TASK RELATIONS
  // =========================================================

  // Organization ↔ Task
  Organization.hasMany(Task, { foreignKey: 'organization_id' });
  Task.belongsTo(Organization, { foreignKey: 'organization_id' });

  // Project ↔ Task
  Project.hasMany(Task, { foreignKey: 'project_id' });
  Task.belongsTo(Project, { foreignKey: 'project_id' });

  // Profile ↔ Task
  Task.belongsTo(Profile, { foreignKey: 'user_id', as: 'owner' });
  Task.belongsTo(Profile, { foreignKey: 'assigned_user_id', as: 'assignedUser' });
  Task.belongsTo(Profile, { foreignKey: 'created_by_user_id', as: 'createdBy' });

  Profile.hasMany(Task, { foreignKey: 'user_id', as: 'ownedTasks' });
  Profile.hasMany(Task, { foreignKey: 'assigned_user_id', as: 'assignedTasks' });
  Profile.hasMany(Task, { foreignKey: 'created_by_user_id', as: 'createdTasks' });

  // =========================================================
  // TASK ASSIGNMENTS (MULTI USER)
  // =========================================================

  Task.hasMany(TaskAssignment, { foreignKey: 'task_id' });
  TaskAssignment.belongsTo(Task, { foreignKey: 'task_id' });

  TaskAssignment.belongsTo(Profile, {
    foreignKey: 'user_id',
    as: 'assignee'
  });

  TaskAssignment.belongsTo(Profile, {
    foreignKey: 'assigned_by',
    as: 'assigner'
  });

  Profile.hasMany(TaskAssignment, {
    foreignKey: 'user_id',
    as: 'taskAssignments'
  });

  Profile.hasMany(TaskAssignment, {
    foreignKey: 'assigned_by',
    as: 'assignedByMe'
  });

  // =========================================================
  // TASK NOTES
  // =========================================================

  Task.hasMany(TaskNote, { foreignKey: 'task_id' });
  TaskNote.belongsTo(Task, { foreignKey: 'task_id' });

  Organization.hasMany(TaskNote, { foreignKey: 'organization_id' });
  TaskNote.belongsTo(Organization, { foreignKey: 'organization_id' });

  Profile.hasMany(TaskNote, { foreignKey: 'user_id' });
  TaskNote.belongsTo(Profile, {
    foreignKey: 'user_id',
    as: 'author'
  });

  // =========================================================
  // TASK URLS
  // =========================================================

  Task.hasMany(TaskURL, { foreignKey: 'task_id' });
  TaskURL.belongsTo(Task, { foreignKey: 'task_id' });

  Organization.hasMany(TaskURL, { foreignKey: 'organization_id' });
  TaskURL.belongsTo(Organization, { foreignKey: 'organization_id' });

  Profile.hasMany(TaskURL, { foreignKey: 'created_by' });
  TaskURL.belongsTo(Profile, {
    foreignKey: 'created_by',
    as: 'creator'
  });

  // =========================================================
  // TASK ATTACHMENTS
  // =========================================================

  Task.hasMany(TaskAttachment, { foreignKey: 'task_id' });
  TaskAttachment.belongsTo(Task, { foreignKey: 'task_id' });

  Organization.hasMany(TaskAttachment, { foreignKey: 'organization_id' });
  TaskAttachment.belongsTo(Organization, { foreignKey: 'organization_id' });

  Profile.hasMany(TaskAttachment, { foreignKey: 'uploaded_by' });
  TaskAttachment.belongsTo(Profile, {
    foreignKey: 'uploaded_by',
    as: 'uploader'
  });

  // =========================================================
  // PROJECT ATTACHMENTS
  // =========================================================

  Organization.hasMany(ProjectAttachment, { foreignKey: 'organization_id' });
  ProjectAttachment.belongsTo(Organization, { foreignKey: 'organization_id' });

  Profile.hasMany(ProjectAttachment, { foreignKey: 'uploaded_by' });
  ProjectAttachment.belongsTo(Profile, {
    foreignKey: 'uploaded_by',
    as: 'uploader'
  });

};
